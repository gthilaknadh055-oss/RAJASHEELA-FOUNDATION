function doGet(e) {
  return ContentService
    .createTextOutput('Rajasheela Foundation Web App (3 PDFs + Payment Confirm) is LIVE!')
    .setMimeType(ContentService.MimeType.TEXT);
}

function saveFileToDrive(base64Data, fileName, mimeType, prefix) {
  if (!base64Data || !fileName) return 'None';
  var folderName = 'Rajasheela Foundation Uploads';
  var folders = DriveApp.getFoldersByName(folderName);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

  var bytes = Utilities.base64Decode(base64Data);
  var blob = Utilities.newBlob(bytes, mimeType || 'application/octet-stream', prefix + '_' + fileName);
  var file = folder.createFile(blob);
  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  return file.getUrl();
}

function ensureHeaders(sheet) {
  var headers = [
    'Timestamp',
    'Application ID',
    'Full Name',
    "Father's Name",
    'Phone Number',
    'GMail',
    'Age',
    'Current Occupation',
    'Social Service Activities',
    'Previous Achievements',
    'Full Address',
    'Supporting PDF',
    'Supporting PDF',
    'Supporting PDF',
    'Payment Status',
    'UTR / Transaction ID',
    'Payment Screenshot Link'
  ];

  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  ensureHeaders(sheet);

  var p = (e && e.parameter) ? e.parameter : {};
  var action = p.action || 'register';

  if (action === 'confirmPayment') {
    var targetAppId = String(p.applicationId || '').trim().toUpperCase();
    var targetPhone = String(p.phone || '').trim();
    var utr = String(p.utrNumber || '').trim();
    var screenshotLink = saveFileToDrive(p.screenshotData, p.screenshotName, p.screenshotType || 'image/png', targetAppId + '_PAY');

    var data = sheet.getDataRange().getValues();
    var matchedRow = -1;

    for (var i = data.length - 1; i >= 1; i--) {
      var rowAppId = String(data[i][1] || '').trim().toUpperCase();
      var rowPhone = String(data[i][4] || '').trim();
      if ((targetAppId && rowAppId === targetAppId) || (targetPhone && rowPhone === targetPhone)) {
        matchedRow = i + 1;
        break;
      }
    }

    if (matchedRow > 1) {
      sheet.getRange(matchedRow, 15, 1, 3).setValues([[
        'UTR Submitted (' + Utilities.formatDate(new Date(), 'Asia/Kolkata', 'dd-MMM-yyyy HH:mm') + ')',
        utr,
        screenshotLink
      ]]);
    } else {
      sheet.appendRow([
        new Date(),
        targetAppId,
        'Payment Only (Unmatched ID)',
        '',
        targetPhone,
        '',
        '',
        '',
        '',
        '',
        '',
        'None',
        'None',
        'None',
        'UTR Submitted (Check ID)',
        utr,
        screenshotLink
      ]);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'payment_confirmed' }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Default: Candidate Registration
  var appId = p.applicationId || '';
  var pdf1Link = saveFileToDrive(p.pdf1Data, p.pdf1Name, 'application/pdf', appId + '_PDF');
  var pdf2Link = saveFileToDrive(p.pdf2Data, p.pdf2Name, 'application/pdf', appId + '_PDF');
  var pdf3Link = saveFileToDrive(p.pdf3Data, p.pdf3Name, 'application/pdf', appId + '_PDF');

  sheet.appendRow([
    new Date(),
    appId,
    p.fullName || '',
    p.fatherName || '',
    p.phone || '',
    p.email || '',
    p.age || '',
    p.occupation || '',
    p.service || '',
    p.achievements || '',
    p.address || '',
    pdf1Link,
    pdf2Link,
    pdf3Link,
    'Pending Payment',
    '',
    ''
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
